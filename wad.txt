#include <stdio.h>

int main() {
    int a[100], n, i, key;
    int low, high, mid, pos = -1;

    printf("Enter number of elements: ");
    if (scanf("%d", &n) != 1 || n < 1 || n > 100) {
        printf("Invalid number of elements (1-100)\n");
        return 1;
    }

    printf("Enter elements in sorted order:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter element to search: ");
    scanf("%d", &key);

    low = 0;
    high = n - 1;

    while (low <= high) {
        mid = (low + high) / 2;

        if (a[mid] == key) {
            pos = mid;
            break;
        }
        else if (key < a[mid])
            high = mid - 1;
        else
            low = mid + 1;
    }

    if (pos != -1)
        printf("Element found at position %d\n", pos + 1);
    else
        printf("Element not found\n");

    return 0;
}

            if (a[j] > a[j + 1]) {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }

        printf("%d ", a[i]);

    printf("\n");

#include <stdio.h>

void quickSort(int a[], int low, int high) {
    int i, j, pivot, temp;

    if (low < high) {
        pivot = a[high];
        i = low - 1;

        for (j = low; j < high; j++) {
            if (a[j] < pivot) {
                i++;
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }

        temp = a[i + 1];
        a[i + 1] = a[high];
        a[high] = temp;

        quickSort(a, low, i);
        quickSort(a, i + 2, high);
    }
}

int main() {
    int a[100], n, i;

    printf("Enter number of elements: ");
    if (scanf("%d", &n) != 1 || n < 1 || n > 100) {
        printf("Invalid number of elements (1-100)\n");
        return 1;
    }

    printf("Enter elements:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    quickSort(a, 0, n - 1);

    printf("Sorted array: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
    return 0;
}

#include <stdio.h>

void merge(int a[], int low, int mid, int high) {
    int temp[100];
    int i = low, j = mid + 1, k = 0;

    while (i <= mid && j <= high) {
        if (a[i] <= a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    while (i <= mid)
        temp[k++] = a[i++];

    while (j <= high)
        temp[k++] = a[j++];

    for (i = low, k = 0; i <= high; i++, k++)
        a[i] = temp[k];
}

void mergeSort(int a[], int low, int high) {
    int mid;

    if (low < high) {
        mid = (low + high) / 2;

        mergeSort(a, low, mid);
        mergeSort(a, mid + 1, high);
        merge(a, low, mid, high);
    }
}

int main() {
    int a[100], n, i;

    printf("Enter number of elements: ");
    if (scanf("%d", &n) != 1 || n < 1 || n > 100) {
        printf("Invalid number of elements (1-100)\n");
        return 1;
    }

    printf("Enter elements:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Sorted array: ");
    for (i = 0; i < n; i++)

    printf("\n");
    return 0;
}

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *head = NULL;
int count = 0;

void insertBeginning(int value) {
    struct Node *newNode;
    newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = value;
    newNode->next = head;
    head = newNode;
    count++;
}

void insertEnd(int value) {
    struct Node *newNode, *temp;
    newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = value;
    newNode->next = NULL;

    if (head == NULL)
        head = newNode;
    else {
        temp = head;
        while (temp->next != NULL)
            temp = temp->next;
        temp->next = newNode;
    }

    count++;
}

void insertPosition(int value, int pos) {
    struct Node *newNode, *temp;
    int i;

    if (pos < 1 || pos > count + 1) {
        printf("Invalid position. Valid range is 1 to %d\n", count + 1);
        return;
    }

    newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = value;

    if (pos == 1) {
        newNode->next = head;
        head = newNode;
    }
    else {
        temp = head;

        for (i = 1; i < pos - 1; i++)
            temp = temp->next;

        newNode->next = temp->next;
        temp->next = newNode;
    }

    count++;
}

void deleteBeginning() {
    struct Node *temp;

    if (head == NULL)
        return;

    temp = head;
    head = head->next;
    free(temp);
    count--;
}

void deleteEnd() {
    struct Node *temp, *prev = NULL;

    if (head == NULL)
        return;

    if (head->next == NULL) {
        free(head);
        head = NULL;
        count--;
        return;
    }

    temp = head;

    while (temp->next != NULL) {
        prev = temp;
        temp = temp->next;
    }

    prev->next = NULL;
    free(temp);
    count--;
}

void deletePosition(int pos) {
    struct Node *temp, *del;
    int i;

    if (head == NULL)
        return;

    if (pos < 1 || pos > count) {
        printf("Invalid position. Valid range is 1 to %d\n", count);
        return;
    }

    if (pos == 1) {
        temp = head;
        head = head->next;
        free(temp);
        count--;
        return;
    }

    temp = head;

    for (i = 1; i < pos - 1; i++)
        temp = temp->next;

    del = temp->next;
    temp->next = del->next;
    free(del);
    count--;
}

void display() {
    struct Node *temp = head;

    printf("\nLinked List: ");

    while (temp != NULL) {
        printf("%d -> ", temp->data);
        temp = temp->next;
    }

    printf("NULL");
    printf("\nNumber of Nodes = %d\n", count);
}


void search(int value) {
    struct Node *temp = head;
    int pos = 1;

    while (temp != NULL) {
        if (temp->data == value) {
            printf("%d found at position %d\n", value, pos);
            return;
        }
        temp = temp->next;
        pos++;
    }
    printf("%d not found in the list.\n", value);
}

void freeList() {
    struct Node *temp;

    while (head != NULL) {
        temp = head;
        head = head->next;
        free(temp);
    }
    count = 0;
}

int main() {
    int n, i, value, pos;

    printf("Enter number of initial nodes: ");
    scanf("%d", &n);

    for (i = 1; i <= n; i++) {
        printf("Enter value for node %d: ", i);
        scanf("%d", &value);
        insertEnd(value);
    }

    display();

    printf("\nEnter value to insert at beginning: ");
    scanf("%d", &value);
    insertBeginning(value);
    display();

    printf("\nEnter value to insert at end: ");
    scanf("%d", &value);
    insertEnd(value);
    display();

    printf("\nEnter value and position to insert: ");
    scanf("%d %d", &value, &pos);
    insertPosition(value, pos);
    display();

    deleteBeginning();
    printf("\nAfter deleting from beginning:");
    display();

    deleteEnd();
    printf("\nAfter deleting from end:");
    display();

    printf("\nEnter position to delete: ");
    scanf("%d", &pos);
    deletePosition(pos);

    printf("After deleting from position %d:", pos);
    display();


    printf("\nEnter value to search: ");
    scanf("%d", &value);
    search(value);

    freeList();
    return 0;
}

    struct Node *next;

    struct Node *newNode;

    newNode = (struct Node*)malloc(sizeof(struct Node));

    newNode->prev = NULL;
    newNode->next = head;

    if (head != NULL)
        head->prev = newNode;

    head = newNode;
    count++;
}

void insertEnd(int value) {
    struct Node *newNode, *temp;

    newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = value;
    newNode->next = NULL;

    if (head == NULL) {
        newNode->prev = NULL;
        head = newNode;
    }
    else {
        temp = head;

        while (temp->next != NULL)
            temp = temp->next;

        temp->next = newNode;
        newNode->prev = temp;
    }

}

    int i;

    newNode = (struct Node*)malloc(sizeof(struct Node));

        head = newNode;

        temp = head;

        for (i = 1; i < pos - 1; i++)
            temp = temp->next;

        newNode->next = temp->next;
        newNode->prev = temp;

        if (temp->next != NULL)
            temp->next->prev = newNode;

        temp->next = newNode;

    count--;

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *head = NULL;
int count = 0;

void insertBeginning(int value) {
    struct Node *newNode, *last;

    newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = value;

    if (head == NULL) {
        newNode->next = newNode;
        head = newNode;
    }
    else {
        last = head;

        while (last->next != head)
            last = last->next;

        newNode->next = head;
        last->next = newNode;
        head = newNode;
    }

    count++;
}

void insertEnd(int value) {
    struct Node *newNode, *last;

    if (head == NULL) {
        insertBeginning(value);
        return;
    }

    newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = value;

    last = head;

    while (last->next != head)
        last = last->next;

    newNode->next = head;
    last->next = newNode;

    count++;
}

void insertPosition(int value, int pos) {
    struct Node *newNode, *temp;
    int i;

    if (pos < 1 || pos > count + 1) {
        printf("Invalid position. Valid range is 1 to %d\n", count + 1);
        return;
    }

    if (pos == 1) {
        insertBeginning(value);
        return;
    }

    newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = value;

    temp = head;

    for (i = 1; i < pos - 1; i++)
        temp = temp->next;

    newNode->next = temp->next;
    temp->next = newNode;

    count++;
}

void deleteBeginning() {
    struct Node *last, *temp;

    if (head == NULL)
        return;

    if (head->next == head) {
        free(head);
        head = NULL;
        count--;
        return;
    }

    last = head;

    while (last->next != head)
        last = last->next;

    temp = head;
    head = head->next;
    last->next = head;

    free(temp);
    count--;
}

void deleteEnd() {
    struct Node *temp, *prev;

    if (head == NULL)
        return;

    if (head->next == head) {
        free(head);
        head = NULL;
        count--;
        return;
    }

    temp = head;
    prev = NULL;

    while (temp->next != head) {
        prev = temp;
        temp = temp->next;
    }

    prev->next = head;

    free(temp);
    count--;
}

void deletePosition(int pos) {
    struct Node *temp, *del;
    int i;

    if (head == NULL)
        return;

    if (pos < 1 || pos > count) {
        printf("Invalid position. Valid range is 1 to %d\n", count);
        return;
    }

    if (pos == 1) {
        deleteBeginning();
        return;
    }

    temp = head;

    for (i = 1; i < pos - 1; i++)
        temp = temp->next;

    del = temp->next;
    temp->next = del->next;

    free(del);
    count--;
}

void display() {
    struct Node *temp = head;

    printf("\nCircular List: ");

    if (temp == NULL) {
        printf("NULL\nNumber of Nodes = %d\n", count);
        return;
    }

    do {
        printf("%d -> ", temp->data);
        temp = temp->next;
    } while (temp != head);

    printf("(head)");
    printf("\nNumber of Nodes = %d\n", count);

void search(int value) {

    if (head == NULL) {

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *next;
};

struct Node *top = NULL;

void push()
{
    struct Node *newNode;
    int value;

    newNode = (struct Node *)malloc(sizeof(struct Node));

    printf("Enter value: ");
    scanf("%d", &value);

    newNode->data = value;
    newNode->next = top;
    top = newNode;

    printf("Element pushed successfully.\n");
}

void pop()
{
    struct Node *temp;

    if (top == NULL)
    {
        printf("Stack Underflow.\n");
        return;
    }

    temp = top;
    printf("Popped element: %d\n", top->data);
    top = top->next;
    free(temp);
}

void peek()
{
    if (top == NULL)
    {
        printf("Stack is empty.\n");
        return;
    }

    printf("Top element: %d\n", top->data);
}

void display()
{
    struct Node *temp = top;

    if (top == NULL)
    {
        printf("Stack is empty.\n");
        return;
    }

    printf("Stack elements:\n");

    while (temp != NULL)
    {
        printf("%d\n", temp->data);
        temp = temp->next;
    }
}


void search() {
    struct Node *temp = top;
    int value, pos = 1;

    printf("Enter value to search: ");
    scanf("%d", &value);

    while (temp != NULL) {
        if (temp->data == value) {
            printf("%d found at position %d from the top.\n", value, pos);
            return;
        }
        temp = temp->next;
        pos++;
    }
    printf("%d not found in the stack.\n", value);
}

void size() {
    struct Node *temp = top;
    int count = 0;

    while (temp != NULL) {
        count++;
        temp = temp->next;
    }
    printf("Number of elements = %d\n", count);
}

int main()
{
    int choice, n, i;
    struct Node *temp;

    do
    {
        printf("\nSTACK\n");
        printf("1. Push\n");
        printf("2. Pop\n");
        printf("3. Peek\n");
        printf("4. Display\n");
        printf("5. Search\n");
        printf("6. Size\n");
        printf("7. Exit\n");

        printf("Enter your choice: ");
        if (scanf("%d", &choice) != 1)
        {
            printf("Invalid input. Exiting...\n");
            break;
        }

        switch (choice)
        {
            case 1:
                printf("Enter number of elements to push: ");
                scanf("%d", &n);

                for (i = 0; i < n; i++)
                {
                    printf("\nElement %d:\n", i + 1);
                    push();
                }

                printf("\n");
                display();
                break;

            case 2:
                pop();
                break;

            case 3:
                peek();
                break;

            case 4:
                display();
                break;

            case 5:
                search();
                break;

            case 6:
                size();

INFIX TO POSTFIX

```
#include <stdio.h> 
#include <ctype.h> 
#include <string.h> 
 
#define MAX 100 
 
char stack[MAX]; 
int top = -1; 
 
void push(char x) { 
    stack[++top] = x; 
} 
 
char pop() { 
    return stack[top--]; 
} 
 
char peek() { 
    return stack[top]; 
} 
 
int precedence(char op) { 
    if (op == '^') 
        return 3; 
    else if (op == '*' || op == '/' || op == '%') 
        return 2; 
    else if (op == '+' || op == '-') 
        return 1; 
    else 
        return 0; 
} 
 
int infixToPostfix(char infix[], char postfix[]) { 
    int i, j = 0; 
    char ch; 

    top = -1;
 
    for (i = 0; infix[i] != '\0'; i++) { 
        ch = infix[i]; 
 
        if (isalnum((unsigned char)ch)) { 
            postfix[j++] = ch; 
        } 
        else if (ch == '(') { 
            push(ch); 
        } 
        else if (ch == ')') { 
            while (top != -1 && peek() != '(') 
                postfix[j++] = pop(); 

            if (top == -1) {
                printf("Error: unbalanced brackets\n");
                return 0;
            }
 
            pop(); 
        } 
        else if (precedence(ch) > 0) { 
            /* '^' is right-associative, all other operators are left-associative */
            while (top != -1 && peek() != '(' &&
                   (precedence(peek()) > precedence(ch) ||
                    (precedence(peek()) == precedence(ch) && ch != '^'))) { 
                postfix[j++] = pop(); 
            } 
 
            push(ch); 
        } 
        else {
            printf("Error: invalid character '%c'\n", ch);
            return 0;
        }
    } 
 
    while (top != -1) {
        if (peek() == '(') {
            printf("Error: unbalanced brackets\n");
            return 0;
        }
        postfix[j++] = pop(); 
    }
 
    postfix[j] = '\0'; 
    return 1;
} 
 
int main() { 
    char infix[MAX], postfix[MAX]; 
 
    printf("Enter infix expression: "); 
    if (scanf("%99s", infix) != 1) {
        printf("Invalid input\n");
        return 1;
    }
 
    if (infixToPostfix(infix, postfix)) 
        printf("Postfix expression: %s\n", postfix); 
 
    return 0; 
}

    {
        front = newNode;
        rear = newNode;
    }
    else
    {
        rear->next = newNode;
        rear = newNode;

int g[MAX][MAX], n;

int valid(int v) {

        return;
    }

        printf("Invalid edge count\n");
        return;

            return;

    int u, v, w = 0;

            return;
        }
    } else {
        printf("u v: ");
        if (scanf("%d%d", &u, &v) != 2) {
            printf("Invalid input\n");
            return;
        }
    }
    if (!valid(u) || !valid(v) || u == v || (op == 1 && w < 0)) {
        printf("Invalid input\n");
        return;
    }
    if (op == 1) {
        g[u][v] = g[v][u] = w;
        printf("Added/updated\n");
    } else if (op == 0) {
        g[u][v] = g[v][u] = INF;
        printf("Removed if present\n");
    } else if (g[u][v] == INF) {
        printf("Edge absent\n");
    } else {
        printf("Weight = %d\n", g[u][v]);
    }
}

void display(void) {
    int i, j;
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++)

void prim(int s) {
    int key[MAX], par[MAX], used[MAX] = {0}, i, j, u, total = 0;
    for (i = 0; i < n; i++) {
        key[i] = INF;
        par[i] = -1;
    }
    key[s] = 0;
    for (i = 0; i < n; i++) {
        u = -1;
        for (j = 0; j < n; j++)
            if (!used[j] && (u == -1 || key[j] < key[u]))
                u = j;

        if (u == -1 || key[u] == INF) {
            printf("Graph disconnected: no MST for all vertices\n");
            return;
        }
        used[u] = 1;
        if (par[u] != -1) {
            printf("%d -- %d : %d\n", par[u], u, key[u]);
            total += key[u];
        }
        for (j = 0; j < n; j++)
            if (!used[j] && g[u][j] != INF && g[u][j] < key[j]) {
                key[j] = g[u][j];
                par[j] = u;
            }
    }
    printf("Total MST cost = %d\n", total);
}

/* g = original graph (direct edges), d = shortest-distance matrix */
int g[MAX][MAX];
long long d[MAX][MAX];

void setEdge(int u, int v, int w, int keepMin) {
    if (u == v) {
        if (w < 0) g[u][u] = w;
        else printf("Self-loop ignored\n");
        return;
    }
    if (!keepMin || w < g[u][v])
        g[u][v] = w;
}

        return;
    }

        for (j = 0; j < n; j++)
            g[i][j] = (i == j ? 0 : INF);
    done = 0;
    neg = 0;

    printf("Directed edges: ");
    if (scanf("%d", &e) != 1 || e < 0) {
        printf("Invalid edge count\n");
        return;
    }

    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            d[i][j] = g[i][j];

    for (k = 0; k < n; k++)
        for (i = 0; i < n; i++)
            for (j = 0; j < n; j++)
                if (d[i][k] < INF && d[k][j] < INF && d[i][k] + d[k][j] < d[i][j]) {
                    d[i][j] = d[i][k] + d[k][j];
                    if (d[i][j] < -INF) d[i][j] = -INF;   /* avoid overflow */
                }

    done = 1;
    neg = 0;
    for (i = 0; i < n; i++)
        if (d[i][i] < 0)
            neg = 1;

        printf("Negative-weight cycle detected: distances are not reliable\n");

#include <stdio.h>
#include <stdlib.h>
#define SIZE 11

typedef struct Node {
    int key;
    struct Node *next;
} Node;

Node *t[SIZE];

int hash(int k) {
    return k % SIZE;
}

int search(int k) {
    Node *p = t[hash(k)];
    while (p) {
        if (p->key == k) return 1;
        p = p->next;
    }
    return 0;
}

void insert(int k) {
    int i = hash(k);
    Node *p;
    if (search(k)) {
        printf("Duplicate key\n");
        return;
    }
    p = malloc(sizeof(Node));
    if (!p) {
        printf("Allocation failed\n");
        return;
    }
    p->key = k;
    p->next = t[i];
    t[i] = p;
    printf("Inserted in bucket %d\n", i);
}

void removeKey(int k) {
    int i = hash(k);
    Node *p = t[i], *prev = NULL;
    while (p && p->key != k) {
        prev = p;
        p = p->next;
    }
    if (!p) {
        printf("Key not found\n");
        return;
    }
    if (prev) prev->next = p->next;
    else t[i] = p->next;
    free(p);
    printf("Deleted\n");
}

void display(void) {
    int i;
    Node *p;
    for (i = 0; i < SIZE; i++) {
        printf("%d: ", i);
        p = t[i];
        while (p) {
            printf("%d -> ", p->key);
            p = p->next;
        }
        printf("NULL\n");
    }
}

void clear(void) {
    int i;
    Node *p, *q;
    for (i = 0; i < SIZE; i++) {
        p = t[i];
        while (p) {
            q = p->next;
            free(p);
            p = q;
        }
        t[i] = NULL;
    }
}

        if (parent[u] != -1) {
            printf("%d - %d\t%d\n", parent[u], u, g[parent[u]][u]);

        }

                key[v] = g[u][v];
                parent[v] = u;

#include <stdio.h>
#define EMPTY -1
#define MAX 50
int table[MAX], size;
int hash(int key) { return key % size; }
int find(int key) {
    int i, idx = hash(key);
    for (i = 0; i < size; i++) {
        int pos = (idx + i) % size;
        if (table[pos] == key) return pos;
        if (table[pos] == EMPTY) break;
    }
    return -1;
}
void insert(int key) {
    int i, idx = hash(key);
    if (find(key) >= 0) {
        printf("Duplicate key %d ignored.\n", key);
        return;
    }
    for (i = 0; i < size; i++) {
        int pos = (idx + i) % size;   /* wrap around */
        if (table[pos] == EMPTY) {
            table[pos] = key;
            printf("Key %d: hash index %d -> placed at index %d\n", key, idx, pos);
            return;
        }
    }
    printf("Table is full. Cannot insert %d\n", key);
}
void search(int key) {
    int pos = find(key);

    else printf("Key %d not found\n", key);
}

    int i;
    printf("\nIndex\tKey\n");
    for (i = 0; i < size; i++) {
        if (table[i] == EMPTY) printf("%d\t--\n", i);

    }
}

            return 1;

            printf("Invalid key. Use non-negative integers.\n");

            continue;
        }

#include <stdio.h>
#include <stdlib.h>
#define MAX 1000
struct Node {
    int key;
    struct Node *next;
};
struct Node *table[MAX];
int m;
int hashFunction(int key) {
    return key % m;
}
int searchKey(int key) {
    int index = hashFunction(key);
    struct Node *current = table[index];
    while (current != NULL) {
        if (current->key == key)
            return 1;
        current = current->next;
    }
    return 0;
}
void insertKey(int key) {
    int index = hashFunction(key);
    struct Node *newNode;
    struct Node *current;
    if (searchKey(key)) {
        printf("Duplicate key %d ignored.\n", key);
        return;
    }
    newNode = malloc(sizeof(struct Node));
    if (newNode == NULL) {
        printf("Memory allocation failed.\n");
        return;
    }
    newNode->key = key;
    newNode->next = NULL;
    if (table[index] == NULL) {
        table[index] = newNode;
    } else {
        current = table[index];
        while (current->next != NULL)
            current = current->next;
        current->next = newNode;
    }
    printf("Inserted %d at bucket %d.\n", key, index);
}
void deleteKey(int key) {
    int index = hashFunction(key);
    struct Node *current = table[index];
    struct Node *previous = NULL;
    while (current != NULL && current->key != key) {
        previous = current;
        current = current->next;
    }
    if (current == NULL) {
        printf("Key %d not found.\n", key);
        return;
    }
    if (previous == NULL)
        table[index] = current->next;
    else
        previous->next = current->next;
    free(current);
    printf("Deleted %d.\n", key);
}
void display(void) {
    int i;
    struct Node *current;
    printf("\nHash Table Using Separate Chaining\n");
    for (i = 0; i < m; i++) {
        printf("Bucket %d: ", i);
        current = table[i];
        while (current != NULL) {
            printf("%d -> ", current->key);
            current = current->next;
        }
        printf("NULL\n");
    }
}
void freeTable(void) {
    int i;
    struct Node *current, *temp;
    for (i = 0; i < m; i++) {
        current = table[i];
        while (current != NULL) {
            temp = current;
            current = current->next;
            free(temp);
        }
        table[i] = NULL;
    }
}
int main(void) {
    int i, n, key, choice;
    printf("Enter hash table size (1-%d): ", MAX);
    if (scanf("%d", &m) != 1 || m < 1 || m > MAX) {
        printf("Invalid table size.\n");
        return 1;
    }
    for (i = 0; i < m; i++)
        table[i] = NULL;
    printf("Enter number of keys: ");
    if (scanf("%d", &n) != 1 || n < 0) {
        printf("Invalid number of keys.\n");
        return 1;
    }
    for (i = 0; i < n; i++) {
        printf("Enter key: ");
        if (scanf("%d", &key) != 1 || key < 0) {
            printf("Invalid key. Use non-negative integers.\n");
            freeTable();
            return 1;
        }
        insertKey(key);
    }
    display();
    do {
        printf("\n1. Insert\n2. Search\n3. Delete\n");
        printf("4. Display\n5. Exit\n");
        printf("Enter choice: ");
        if (scanf("%d", &choice) != 1)
            break;
        switch (choice) {
            case 1:
                printf("Enter key to insert: ");
                if (scanf("%d", &key) != 1 || key < 0) {
                    printf("Invalid key.\n");
                    freeTable();
                    return 1;
                }
                insertKey(key);
                break;
            case 2:
                printf("Enter key to search: ");
                if (scanf("%d", &key) != 1 || key < 0) {
                    printf("Invalid key.\n");
                    freeTable();
                    return 1;
                }
                if (searchKey(key))
                    printf("Key found.\n");
                else
                    printf("Key not found.\n");
                break;
            case 3:
                printf("Enter key to delete: ");
                if (scanf("%d", &key) != 1 || key < 0) {
                    printf("Invalid key.\n");
                    freeTable();
                    return 1;
                }
                deleteKey(key);
                break;
            case 4:
                display();
                break;
            case 5:
                printf("Exiting program.\n");
                break;
            default:
                printf("Invalid choice.\n");
        }
    } while (choice != 5);
    freeTable();
    return 0;
}

21. HEAP (MAX HEAP)

```
#include <stdio.h>

#define MAX 100

int heap[MAX + 1];   /* 1-indexed: parent of i is i/2, children are 2i and 2i+1 */
int n = 0;

void swap(int *x, int *y) {
    int temp = *x;
    *x = *y;
    *y = temp;
}

void heapifyDown(int i) {
    int largest, l, r;

    while (1) {
        largest = i;
        l = 2 * i;
        r = 2 * i + 1;

        if (l <= n && heap[l] > heap[largest])
            largest = l;

        if (r <= n && heap[r] > heap[largest])
            largest = r;

        if (largest == i)
            break;

        swap(&heap[i], &heap[largest]);
        i = largest;
    }
}

void insert()
{
    int value, i;

    if (n == MAX)
    {
        printf("Heap Overflow.\n");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &value);

    n++;
    heap[n] = value;
    i = n;

    while (i > 1 && heap[i / 2] < heap[i])
    {
        swap(&heap[i / 2], &heap[i]);
        i = i / 2;
    }

    printf("Element inserted successfully.\n");
}

void deleteMax()
{
    if (n == 0)
    {
        printf("Heap Underflow.\n");
        return;
    }

    printf("Deleted maximum element: %d\n", heap[1]);

    heap[1] = heap[n];
    n--;

    heapifyDown(1);
}

void peek()
{
    if (n == 0)
    {
        printf("Heap is empty.\n");
        return;
    }

    printf("Maximum element: %d\n", heap[1]);
}

void display()
{
    int i;

    if (n == 0)
    {
        printf("Heap is empty.\n");
        return;
    }

    printf("Heap elements (array order): ");

    for (i = 1; i <= n; i++)
        printf("%d ", heap[i]);

    printf("\n");
}

void search() {
    int value, i;

    printf("Enter value to search: ");
    scanf("%d", &value);

    for (i = 1; i <= n; i++) {
        if (heap[i] == value) {
            printf("%d found at index %d.\n", value, i);
            return;
        }
    }
    printf("%d not found in the heap.\n", value);
}

void size() {
    printf("Number of elements = %d\n", n);
}

int main()
{
    int choice, k, i;

    do
    {
        printf("\nMAX HEAP\n");
        printf("1. Insert\n");
        printf("2. Delete maximum\n");
        printf("3. Peek\n");
        printf("4. Display\n");
        printf("5. Search\n");
        printf("6. Size\n");
        printf("7. Exit\n");

        printf("Enter your choice: ");
        if (scanf("%d", &choice) != 1)
        {
            printf("Invalid input. Exiting...\n");
            break;
        }

        switch (choice)
        {
            case 1:
                printf("Enter number of elements to insert: ");
                scanf("%d", &k);

                for (i = 0; i < k; i++)
                {
                    printf("\nElement %d:\n", i + 1);
                    insert();
                }

                printf("\n");
                display();
                break;

            case 2:
                deleteMax();
                break;

            case 3:
                peek();
                break;

            case 4:
                display();
                break;

            case 5:
                search();
                break;

            case 6:
                size();
                break;

            case 7:
                printf("Exiting...\n");
                break;

            default:
                printf("Invalid choice.\n");
        }

    } while (choice != 7);

    return 0;
}

```
#include <stdio.h>

void heapify(int a[], int n, int i) {
    int largest = i, l = 2 * i + 1, r = 2 * i + 2, temp;

    if (l < n && a[l] > a[largest])
        largest = l;

    if (r < n && a[r] > a[largest])
        largest = r;

    if (largest != i) {
        temp = a[i];
        a[i] = a[largest];
        a[largest] = temp;

        heapify(a, n, largest);
    }
}

void heapSort(int a[], int n) {
    int i, temp;

    for (i = n / 2 - 1; i >= 0; i--)
        heapify(a, n, i);

    for (i = n - 1; i > 0; i--) {
        temp = a[0];
        a[0] = a[i];
        a[i] = temp;

        heapify(a, i, 0);
    }
}

int main() {
    int a[100], n, i;

    printf("Enter number of elements: ");
    if (scanf("%d", &n) != 1 || n < 1 || n > 100) {
        printf("Invalid number of elements (1-100)\n");
        return 1;
    }

    printf("Enter elements:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    heapSort(a, n);

    printf("Sorted array: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
    return 0;
}
